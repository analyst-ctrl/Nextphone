PASO 3: CREAR COLUMNAS CALCULADAS (en Datos_Limpios)
================================================================================
  Columna K - "PRIORIDAD":
    =IF(OR(F2>50, E2>3), "ALTA", IF(OR(F2>20, E2>1), "MEDIA", "BAJA"))

  Columna L - "TIENE_TELEFONO":
    =IF(OR(I2<>"", J2<>""), "SI", "NO")

  Columna M - "TIENE_CORREO":
    =IF(H2<>"", "SI", "NO")

  Columna N - "TIENE_CONTACTO":
    =IF(G2<>"", "SI", "NO")

  Columna O - "MRC_POR_RGU":
    =IF(E2>0, F2/E2, F2)

  Columna P - "CATEGORIA_MRC":
    =IF(F2>=100, "VIP", IF(F2>=50, "PREMIUM", IF(F2>=20, "ESTANDAR", "BASICO")))

  Columna Q - "TIPO_CLIENTE":
    =IF(B2="SIN NOMBRE", "SIN IDENTIFICAR", "IDENTIFICADO")
