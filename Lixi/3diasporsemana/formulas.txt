================================================================================
  5. SQL - ESTRUCTURA DE BASE DE DATOS
================================================================================

  Tablas en iphone_julio_7_24.db / sharep.db / promo_3dias.db:

  -- Dimensión: Supervisores
  CREATE TABLE supervisores (
      id      INTEGER PRIMARY KEY,
      nombre  TEXT NOT NULL UNIQUE
  );

  -- Dimensión: Asesores (vendedores)
  CREATE TABLE asesores (
      id      INTEGER PRIMARY KEY,
      nombre  TEXT NOT NULL UNIQUE
  );

  -- Relación: Supervisor-Asesor (M:N)
  CREATE TABLE supervisor_asesor (
      supervisor_id INTEGER NOT NULL,
      asesor_id     INTEGER NOT NULL,
      PRIMARY KEY (supervisor_id, asesor_id),
      FOREIGN KEY (supervisor_id) REFERENCES supervisores(id),
      FOREIGN KEY (asesor_id) REFERENCES asesores(id)
  );

  -- Dimensión: Modelos de equipo
  CREATE TABLE modelos (
      id      INTEGER PRIMARY KEY,
      nombre  TEXT NOT NULL UNIQUE
  );

  -- Dimensión: Planes vendidos
  CREATE TABLE planes (
      id      INTEGER PRIMARY KEY,
      nombre  TEXT NOT NULL UNIQUE
  );

  -- Dimensión: Status del cliente
  CREATE TABLE status_cliente (
      id      INTEGER PRIMARY KEY,
      nombre  TEXT NOT NULL UNIQUE
  );

  -- Dimensión: Tipo de venta
  CREATE TABLE tipo_venta (
      id      INTEGER PRIMARY KEY,
      nombre  TEXT NOT NULL UNIQUE
  );

  -- Dimensión: Clientes
  CREATE TABLE clientes (
      id              INTEGER PRIMARY KEY,
      nombre          TEXT NOT NULL,
      identificacion  TEXT,
      UNIQUE(nombre, identificacion)
  );

  -- Tabla de hechos: Ventas
  CREATE TABLE ventas (
      id              INTEGER PRIMARY KEY,
      supervisor_id   INTEGER,
      asesor_id       INTEGER,
      modelo_id       INTEGER,
      plan_id         INTEGER,
      status_id       INTEGER,
      tipo_venta_id   INTEGER,
      cliente_id      INTEGER,
      precio_plan     REAL,
      fecha_creacion  TIMESTAMP,
      fecha_modific   TIMESTAMP,
      dia_crea        TEXT,
      dia_mod         TEXT,
      promo           TEXT,
      modo            TEXT,
      -- Llaves foráneas
      FOREIGN KEY (supervisor_id) REFERENCES supervisores(id),
      FOREIGN KEY (asesor_id) REFERENCES asesores(id),
      FOREIGN KEY (modelo_id) REFERENCES modelos(id),
      FOREIGN KEY (plan_id) REFERENCES planes(id),
      FOREIGN KEY (status_id) REFERENCES status_cliente(id),
      FOREIGN KEY (tipo_venta_id) REFERENCES tipo_venta(id),
      FOREIGN KEY (cliente_id) REFERENCES clientes(id)
  );

  -- Vista para consultas fáciles
  CREATE VIEW v_ventas AS
  SELECT
      v.id,
      cl.nombre AS cliente,
      s.nombre AS supervisor,
      a.nombre AS asesor,
      m.nombre AS modelo,
      p.nombre AS plan,
      v.precio_plan,
      st.nombre AS status,
      v.fecha_creacion,
      v.fecha_modific,
      v.dia_crea,
      v.dia_mod,
      v.promo
  FROM ventas v
  LEFT JOIN clientes cl ON v.cliente_id = cl.id
  LEFT JOIN supervisores s ON v.supervisor_id = s.id
  LEFT JOIN asesores a ON v.asesor_id = a.id
  LEFT JOIN modelos m ON v.modelo_id = m.id
  LEFT JOIN planes p ON v.plan_id = p.id
  LEFT JOIN status_cliente st ON v.status_id = st.id;

  6. SQL - VENTAS POR SUPERVISOR
================================================================================

  -- Total de ventas por supervisor (todos los equipos)
  SELECT s.nombre,
         COUNT(*) as ventas,
         COUNT(DISTINCT v.fecha_creacion) as dias_con_venta
  FROM ventas v
  JOIN supervisores s ON v.supervisor_id = s.id
  WHERE strftime('%Y-%m', v.fecha_creacion) = '2026-07'
  GROUP BY s.nombre
  ORDER BY ventas DESC;


================================================================================
  7. SQL - VENTAS IPHONE POR SUPERVISOR
================================================================================

  -- Ventas de iPhone por supervisor (con porcentaje)
  SELECT s.nombre,
         COUNT(*) as total_ventas,
         SUM(CASE WHEN m.nombre LIKE '%IPHONE%' THEN 1 ELSE 0 END) as iPhones,
         ROUND(SUM(CASE WHEN m.nombre LIKE '%IPHONE%' THEN 1 ELSE 0 END)
               * 100.0 / COUNT(*), 1) as pct_iphone,
         COUNT(DISTINCT v.fecha_creacion) as dias_con_venta,
         ROUND(COUNT(*) * 1.0 /
               COUNT(DISTINCT v.fecha_creacion), 1) as promedio_dia
  FROM ventas v
  JOIN supervisores s ON v.supervisor_id = s.id
  JOIN modelos m ON v.modelo_id = m.id
  WHERE strftime('%Y-%m', v.fecha_creacion) = '2026-07'
  GROUP BY s.nombre
  ORDER BY iPhones DESC;


================================================================================
  8. SQL - RANKING DE SUPERVISORES
================================================================================

  -- Ranking completo con posición
  SELECT ROW_NUMBER() OVER (ORDER BY COUNT(*) DESC) as posicion,
         s.nombre as supervisor,
         COUNT(*) as ventas,
         ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (), 1) as pct_del_total,
         COUNT(DISTINCT v.fecha_creacion) as dias,
         ROUND(COUNT(*) * 1.0 /
               COUNT(DISTINCT v.fecha_creacion), 1) as promedio_dia
  FROM ventas v
  JOIN supervisores s ON v.supervisor_id = s.id
  WHERE v.promo = 'SI'
  GROUP BY s.nombre
  ORDER BY ventas DESC;


================================================================================
  9. SQL - ASESORES POR SUPERVISOR
================================================================================

  -- Qué asesores pertenecen a cada supervisor
  SELECT s.nombre as supervisor,
         a.nombre as asesor,
         COUNT(v.id) as ventas
  FROM supervisor_asesor sa
  JOIN supervisores s ON sa.supervisor_id = s.id
  JOIN asesores a ON sa.asesor_id = a.id
  LEFT JOIN ventas v ON v.asesor_id = a.id
     AND strftime('%Y-%m', v.fecha_creacion) = '2026-07'
  GROUP BY s.nombre, a.nombre
  ORDER BY s.nombre, ventas DESC;


================================================================================
  10. SQL - VENTAS POR MODELO
================================================================================

  -- Ranking de modelos de iPhone vendidos
  SELECT m.nombre as modelo,
         COUNT(*) as unidades,
         s.nombre as supervisor
  FROM ventas v
  JOIN modelos m ON v.modelo_id = m.id
  JOIN supervisores s ON v.supervisor_id = s.id
  WHERE m.nombre LIKE '%IPHONE%'
    AND strftime('%Y-%m', v.fecha_creacion) = '2026-07'
  GROUP BY m.nombre, s.nombre
  ORDER BY unidades DESC;

  -- Resumen general para el dashboard HTML
  SELECT
    (SELECT COUNT(*) FROM ventas
     WHERE strftime('%Y-%m', fecha_creacion) = '2026-07') as total_julio,

    (SELECT COUNT(*) FROM ventas v

  16. EXCEL - COMPARATIVA PERIODOS
================================================================================

  Para comparar diferentes periodos:

  Python:
  ---------------------------------------------------------------
  # Julio Completo (todos los días, Promo Si, iPhone)
  SELECT COUNT(*) as total_julio
  FROM ventas v JOIN modelos m ON v.modelo_id = m.id
  WHERE strftime('%Y-%m', v.fecha_creacion) = '2026-07'
    AND v.promo = 'SI'
    AND m.nombre LIKE '%IPHONE%';

  # Julio LMV (solo Lun/Mie/Vie, Promo Si, iPhone)
  SELECT COUNT(*) as total_lmj
  FROM ventas v JOIN modelos m ON v.modelo_id = m.id
  WHERE strftime('%Y-%m', v.fecha_creacion) = '2026-07'
    AND v.promo = 'SI'
    AND m.nombre LIKE '%IPHONE%'
    AND v.dia_crea IN ('Lunes','Miércoles','Viernes');

  # Porcentaje de ventas en días promo
  %_LMV = (total_lmj / total_julio) * 100

  # Ventas fuera de días promo
  otros_dias = total_julio - total_lmj
